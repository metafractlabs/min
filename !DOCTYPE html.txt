  <script>
    let currentSlide = 1;
    const messages = [
      "I love you more than anything in this world.",
      "You are the most handsome man I've ever met.",
      "I feel so lucky to have you in my life.",
      "I still remember how we met in our childhood... it feels like destiny.",
      "I promise I will never leave you, no matter what.",
      "You make me the happiest person alive.",
      "I can't wait to spend the rest of my life with you.",
      "You are my everything. ❤️"
    ];
    let messageIndex = 0;

    function nextSlide() {
      const current = document.getElementById(`slide${currentSlide}`);
      current.classList.remove('active');

      currentSlide++;
      const next = document.getElementById(`slide${currentSlide}`);
      next.classList.add('active');

      if (currentSlide === 2) {
        // Start playing music on the second slide
        const audio = document.getElementById('background-music');
        audio.play();

        // Add click event for messages
        document.body.addEventListener('click', showNextMessage);
      }
    }

    function showNextMessage() {
      const messageElement = document.getElementById('message');
      if (messageIndex < messages.length) {
        messageElement.textContent = messages[messageIndex];
        messageIndex++;
      } else {
        // Move to the final slide
        nextSlide();
        document.body.removeEventListener('click', showNextMessage);
      }
    }
  </script>
